cmake_minimum_required(VERSION 3.21)
project(Wayfinder VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_executable(wayfinder main.cpp)
target_link_libraries(wayfinder PRIVATE Qt6::Widgets)
install(TARGETS wayfinder RUNTIME DESTINATION bin)
install(FILES wayfinder.desktop DESTINATION share/applications)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(wayfinder-desktop-test desktop-test.cpp)
  target_link_libraries(wayfinder-desktop-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME desktop COMMAND wayfinder-desktop-test)
  set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
